-- WARNING: This schema is for context only and is not meant to be run.
-- Table order and constraints may not be valid for execution.

CREATE TABLE public.annotations (
  id text NOT NULL,
  highlight_id text NOT NULL UNIQUE,
  text text NOT NULL,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT annotations_pkey PRIMARY KEY (id),
  CONSTRAINT annotations_highlight_id_fkey FOREIGN KEY (highlight_id) REFERENCES public.highlights(id)
);
CREATE TABLE public.book_requests (
  id text NOT NULL,
  user_id text NOT NULL,
  title text NOT NULL,
  author text,
  reason text,
  status text NOT NULL DEFAULT 'PENDING'::text CHECK (status = ANY (ARRAY['PENDING'::text, 'APPROVED'::text, 'REJECTED'::text, 'ADDED'::text])),
  admin_notes text,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  resolved_at timestamp without time zone,
  CONSTRAINT book_requests_pkey PRIMARY KEY (id),
  CONSTRAINT book_requests_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.book_text (
  book_id text NOT NULL,
  content text NOT NULL,
  word_count integer,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT book_text_pkey PRIMARY KEY (book_id),
  CONSTRAINT book_text_book_id_fkey FOREIGN KEY (book_id) REFERENCES public.books(id)
);
CREATE TABLE public.books (
  id text NOT NULL,
  title text NOT NULL,
  author text NOT NULL,
  domain text,
  subdomain text,
  file_path text,
  is_public integer DEFAULT 1,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT books_pkey PRIMARY KEY (id)
);
CREATE TABLE public.chapters (
  id text NOT NULL,
  book_id text NOT NULL,
  number integer NOT NULL,
  title text NOT NULL,
  start_offset integer NOT NULL,
  end_offset integer NOT NULL,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT chapters_pkey PRIMARY KEY (id),
  CONSTRAINT chapters_book_id_fkey FOREIGN KEY (book_id) REFERENCES public.books(id)
);
CREATE TABLE public.highlights (
  id text NOT NULL,
  user_id text NOT NULL,
  chapter_id text NOT NULL,
  start_offset integer NOT NULL,
  end_offset integer NOT NULL,
  color text DEFAULT 'yellow'::text,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT highlights_pkey PRIMARY KEY (id),
  CONSTRAINT highlights_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id),
  CONSTRAINT highlights_chapter_id_fkey FOREIGN KEY (chapter_id) REFERENCES public.chapters(id)
);
CREATE TABLE public.saved_books (
  id text NOT NULL,
  user_id text NOT NULL,
  book_id text NOT NULL,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT saved_books_pkey PRIMARY KEY (id),
  CONSTRAINT saved_books_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id),
  CONSTRAINT saved_books_book_id_fkey FOREIGN KEY (book_id) REFERENCES public.books(id)
);
CREATE TABLE public.schema_versions (
  version integer NOT NULL,
  applied_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  description text,
  CONSTRAINT schema_versions_pkey PRIMARY KEY (version)
);
CREATE TABLE public.sessions (
  id text NOT NULL,
  user_id text NOT NULL,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  expires_at timestamp without time zone NOT NULL,
  CONSTRAINT sessions_pkey PRIMARY KEY (id),
  CONSTRAINT sessions_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.users (
  id text NOT NULL,
  email text NOT NULL UNIQUE,
  password_hash text NOT NULL,
  display_name text,
  role text NOT NULL DEFAULT 'user'::text CHECK (role = ANY (ARRAY['user'::text, 'admin'::text])),
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT users_pkey PRIMARY KEY (id)
);
